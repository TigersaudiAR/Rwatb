        // تحديث الرتبة والدرجة تلقائيًا
        function updateRankAndDegree() {
            const ranksList = ["جندي", "جندي أول", "عريف", "وكيل رقيب", "رقيب", "رقيب أول", "رئيس رقباء", "ملازم"];
            const oldRank = document.getElementById('old-rank').value;
            
            // تحديد الرتبة التالية
            const oldRankIndex = ranksList.indexOf(oldRank);
            if (oldRankIndex >= 0 && oldRankIndex < ranksList.length - 1) {
                document.getElementById('new-rank').value = ranksList[oldRankIndex + 1];
            }
            
            updateSalaryScale();
        }
        
        // تحديث الدرجة الحالية بناءً على الدرجة السابقة
        function updateBasedOnOldDegree() {
            const oldDegree = parseInt(document.getElementById('old-degree').value);
            const newDegree = Math.max(1, oldDegree - 1); // الدرجة الحالية هي الدرجة السابقة - 1 (لا تقل عن 1)
            document.getElementById('new-degree').value = newDegree.toString();
            
            // البحث عن الراتب المناسب بناءً على الراتب السابق
            findOptimalDegreeBasedOnSalary();
            
            updateSalaryScale();
        }
        
        // البحث عن الدرجة المناسبة للرتبة الجديدة بناءً على الراتب السابق
        function findOptimalDegreeBasedOnSalary() {
            const oldRank = document.getElementById('old-rank').value;
            const newRank = document.getElementById('new-rank').value;
            const oldDegree = document.getElementById('old-degree').value;
            
            // التحقق من وجود العناصر قبل الوصول إليها
            const oldSalaryElement = document.getElementById(`scale-${oldRank}-${oldDegree}`);
            if (!oldSalaryElement) return; // التحقق من وجود العنصر
            
            const oldSalary = Number(oldSalaryElement.value);
            
            // البحث عن أقرب درجة في الرتبة الجديدة بحيث يكون راتبها ليس أقل من الراتب السابق
            let foundDegree = 15; // نبدأ من أعلى درجة
            for (let i = 15; i >= 1; i--) {
                const degreeElement = document.getElementById(`scale-${newRank}-${i}`);
                if (!degreeElement) continue;
                
                const newSalary = Number(degreeElement.value);
                if (newSalary >= oldSalary) {
                    foundDegree = i;
                }
            }
            
            // تحديث الدرجة الحالية
            document.getElementById('new-degree').value = foundDegree.toString();
        }

            const oldRank = document.getElementById('old-rank').value;
            const newRank = document.getElementById('new-rank').value;
            const oldDegree = document.getElementById('old-degree').value;
            const newDegree = document.getElementById('new-degree').value;
            
            // التحقق من وجود العناصر قبل الوصول إليها
            const oldSalaryElement = document.getElementById(`scale-${oldRank}-${oldDegree}`);

            document.getElementById('new-basic-salary').value = newSalary;

                return;

            const oldRankFirstDegreeSalary = Number(oldRankFirstDegreeElement.value) || 0;
            const newRankFirstDegreeSalary = Number(newRankFirstDegreeElement.value) || 0;

            const newTerrorismElement = document.getElementById('new-terrorism');

            }

            const oldSecurityValue = parseFloat((oldBasicSalary * 0.25).toFixed(2));
            const newSecurityValue = parseFloat((newBasicSalary * 0.25).toFixed(2));
            
            const oldSecurityElement = document.getElementById('old-security');
            const newSecurityElement = document.getElementById('new-security');
            
            if (oldSecurityElement && newSecurityElement) {

            const oldRetirementElement = document.getElementById('old-retirement');
            const newRetirementElement = document.getElementById('new-retirement');

        // حساب الفروقات
        function calculateDifferences() {
            const nameInput = document.getElementById('name');
            const generalNumberInput = document.getElementById('general-number');
            const unitInput = document.getElementById('unit');
            
            if (!nameInput || !generalNumberInput || !unitInput) {
                console.error("لم يتم العثور على حقول البيانات الأساسية");
                return;
            }
            
            const name = nameInput.value.trim();
            const generalNumber = generalNumberInput.value.trim();
            const unit = unitInput.value.trim();
            
            if (!name || !generalNumber || !unit) {
                alert('يرجى إدخال الاسم والرقم العام والوحدة');
                return;
            }
            
            // الحصول على تواريخ الحسبة
            const startDateInput = document.getElementById('start-date');
            const endDateInput = document.getElementById('end-date');
            
            if (!startDateInput || !endDateInput) {
                console.error("لم يتم العثور على حقول التواريخ");
                return;
            }
            
            const startDate = new Date(startDateInput.value);
            const endDate = new Date(endDateInput.value);
            
            if (isNaN(startDate.getTime()) || isNaN(endDate.getTime())) {
                alert('يرجى إدخال تاريخ بداية ونهاية الحسبة');
                return;
            }
            
            // حساب عدد الأيام مع مراعاة قاعدة الـ 30 يومًا للشهر
            // التحقق مما إذا كان تاريخ النهاية هو آخر يوم في الشهر
            const isEndDateLastDayOfMonth = function(date) {
                // إنشاء نسخة من التاريخ واختبار ما إذا كان آخر يوم في الشهر
                const testDate = new Date(date.getTime());
                testDate.setDate(testDate.getDate() + 1);
                return testDate.getDate() === 1; // إذا كان اليوم التالي هو الأول من الشهر فهذا يعني أن التاريخ الحالي هو آخر يوم
            };
            
            // حساب الفرق بالأشهر والأيام
            let daysDiff;
            
            if (isEndDateLastDayOfMonth(endDate)) {
                // إذا كان تاريخ النهاية هو آخر يوم في الشهر، نستخدم حساب الأشهر
                
                // حساب عدد الأشهر الكاملة
                let months = (endDate.getFullYear() - startDate.getFullYear()) * 12 + (endDate.getMonth() - startDate.getMonth());
                
                // معالجة الأيام المتبقية في الشهر الأول
                if (startDate.getDate() !== 1) {
                    // نحسب عدد الأيام المتبقية في الشهر الأول
                    const daysInStartMonth = 30;
                    const remainingDaysInStartMonth = daysInStartMonth - startDate.getDate() + 1;
                    
                    // إذا كان هناك أقل من شهر
                    if (months === 0) {
                        daysDiff = endDate.getDate() - startDate.getDate() + 1;
                    } else {
                        // نحسب عدد الأيام في الأشهر الكاملة زائد الأيام المتبقية في الشهر الأول
                        daysDiff = (months - 1) * 30 + remainingDaysInStartMonth + endDate.getDate();
                    }
                } else {
                    // إذا كان تاريخ البداية هو الأول من الشهر، فالحساب يكون أبسط
                    daysDiff = months * 30 + endDate.getDate() - startDate.getDate() + 1;
                }
            } else {
                // في حالة عدم اختيار آخر يوم في الشهر، نستخدم الحساب العادي
                daysDiff = Math.ceil((endDate - startDate) / (1000 * 60 * 60 * 24)) + 1; // إضافة 1 لتضمين اليوم الأخير
            }
            
            // إضافة عرض عدد الأيام المحسوبة وتفاصيل الحساب (للتوضيح)
            console.log(`عدد الأيام المحسوبة: ${daysDiff}`);
            
            // إضافة تفاصيل الحساب للمستخدم
            if (isEndDateLastDayOfMonth(endDate)) {
                const startDateEl = document.getElementById('start-date');
                if (startDateEl && startDateEl.parentNode && startDateEl.parentNode.parentNode) {
                    // حذف أي عنصر معلومات سابق
                    const existingInfo = document.querySelector('.days-info');
                    if (existingInfo) existingInfo.remove();
                    
                    const daysInfo = document.createElement('div');
                    daysInfo.className = 'p-3 bg-blue-50 dark:bg-blue-900/20 rounded-md mt-2 text-sm days-info';
                    daysInfo.innerHTML = `
                        <div class="font-semibold mb-1">تفاصيل حساب الأيام (${daysDiff} يوم):</div>
                        <div>• تاريخ البداية: ${startDate.toLocaleDateString('ar-SA')}</div>
                        <div>• تاريخ النهاية: ${endDate.toLocaleDateString('ar-SA')} (آخر يوم في الشهر)</div>
                        <div>• تم احتساب الشهر بـ 30 يوم حسب النظام</div>
                    `;
                    
                    startDateEl.parentNode.parentNode.appendChild(daysInfo);
                    
                    // حذف العنصر بعد 10 ثوانٍ
                    setTimeout(() => {
                        daysInfo.classList.add('opacity-0', 'transition-opacity', 'duration-500');
                        setTimeout(() => daysInfo.remove(), 500);
                    }, 10000);
                }
            }
            
            // جمع معلومات الرتبة

                console.error("لم يتم العثور على عناصر الرتبة");
                return;
            }
            
            const oldRank = oldRankSelect.value;
            const newRank = newRankSelect.value;
            
            // جمع مفردات الراتب
            const oldBasicSalaryInput = document.getElementById('old-basic-salary');
            const newBasicSalaryInput = document.getElementById('new-basic-salary');
            const oldTerrorismInput = document.getElementById('old-terrorism');
            const newTerrorismInput = document.getElementById('new-terrorism');
            const oldSecurityInput = document.getElementById('old-security');
            const newSecurityInput = document.getElementById('new-security');

                console.error("لم يتم العثور على عناصر مفردات الراتب");
                return;
            }

            const newBasicSalary = Number(newBasicSalaryInput.value) || 0;

            function collectAdditionalAllowances(daysDiff) {
                const additionalAllowances = [];
                const additionalDeductions = [];
                const allowanceRows = document.querySelectorAll('#allowances-table tbody tr[data-allowance-id]');
                
                allowanceRows.forEach(row => {
                    if (!row.cells || row.cells.length < 3) return;
                    
                    const name = row.cells[0].textContent.trim();
                    const inputs = row.querySelectorAll('input');
                    
                    if (inputs.length < 2) return;
                    
                    const oldAmount = Number(inputs[0].value) || 0;
                    const newAmount = Number(inputs[1].value) || 0;
                    
                    // إذا كان البدل الحالي 0 والسابق له قيمة، نضعه في الحسميات
                    if (newAmount === 0 && oldAmount > 0) {
                        additionalDeductions.push({
                            type: name,
                            oldAmount,
                            newAmount,
                            days: daysDiff,
                            diffAmount: Math.round(((newAmount - oldAmount) / 30) * daysDiff)
                        });
                    } 
                    // إذا كان هناك فرق إيجابي أو سلبي، نضعه في البدلات
                    else if (newAmount !== oldAmount) {
                        additionalAllowances.push({
                            type: name,
                            oldAmount,
                            newAmount,
                            days: daysDiff,
                            diffAmount: Math.round(((newAmount - oldAmount) / 30) * daysDiff)
                        });
                    }
                });
                
                return { additionalAllowances, additionalDeductions };
            }
            
            // جمع البدلات الإضافية
            const { additionalAllowances, additionalDeductions } = collectAdditionalAllowances(daysDiff);
            
            // حساب الفروقات
            const basicDiff = newBasicSalary - oldBasicSalary;
            const terrorismDiff = newTerrorism - oldTerrorism;
            const securityDiff = newSecurity - oldSecurity;
            
            // حساب مبالغ الفروقات باستخدام 30 يوم لكل شهر مع دعم الأرقام العشرية
            const basicDiffAmount = parseFloat(((basicDiff / 30) * daysDiff).toFixed(2));
            const terrorismDiffAmount = parseFloat(((terrorismDiff / 30) * daysDiff).toFixed(2));
            const securityDiffAmount = parseFloat(((securityDiff / 30) * daysDiff).toFixed(2));
            
            // حساب فروقات الحسميات باستخدام الأرقام العشرية
            const retirementDiff = newRetirement - oldRetirement;
            const retirementDiffAmount = parseFloat(((retirementDiff / 30) * daysDiff).toFixed(2));
            
            // إجمالي الفروقات
            let totalPositiveDiff = basicDiffAmount + terrorismDiffAmount + securityDiffAmount;
            let totalNegativeDiff = retirementDiffAmount > 0 ? retirementDiffAmount : 0;
            
            // إضافة البدلات الإضافية
            for (const allowance of additionalAllowances) {
                if (allowance.diffAmount > 0) {
                    totalPositiveDiff += allowance.diffAmount;
                } else if (allowance.diffAmount < 0) {
                    totalNegativeDiff += Math.abs(allowance.diffAmount);
                }
            }
            
            // إضافة الحسميات الإضافية
            for (const deduction of additionalDeductions) {
                totalNegativeDiff += Math.abs(deduction.diffAmount);
            }
            
            const netDiff = totalPositiveDiff - totalNegativeDiff;
            
            differenceTotal = netDiff;
            
            // إنشاء قائمة الفروقات
            const differences = [
                { type: "الراتب الأساسي", oldAmount: oldBasicSalary, newAmount: newBasicSalary, days: daysDiff, diffAmount: basicDiffAmount },
                { type: "بدل مكافحة الإرهاب", oldAmount: oldTerrorism, newAmount: newTerrorism, days: daysDiff, diffAmount: terrorismDiffAmount },
                { type: "علاوة الأمن (شرطة عسكرية)", oldAmount: oldSecurity, newAmount: newSecurity, days: daysDiff, diffAmount: securityDiffAmount },
                ...additionalAllowances
            ];
            
            // إنشاء قائمة الحسميات
            const deductions = [
                { type: "التقاعد", oldAmount: oldRetirement, newAmount: newRetirement, days: daysDiff, diffAmount: retirementDiffAmount },
                ...additionalDeductions
            ];
            
            // تخزين النتائج للعرض لاحقًا
            calculatedData = {
                name,
                generalNumber,
                idNumber: document.getElementById('id-number')?.value.trim() || '',
                unit,
                commencementDate: document.getElementById('commencement-date')?.value || '',
                startDate: startDateInput.value,
                endDate: endDateInput.value,
                oldRank,
                newRank,
                oldDegree: document.getElementById('old-degree')?.value || '1',
                newDegree: document.getElementById('new-degree')?.value || '1',
                daysDiff,
                differences,
                deductions,
                totalPositiveDiff,
                totalNegativeDiff,
                netDiff
            };

            // تبديل إلى نموذج فروقات الترقية
            switchTab('forms');
            previewDifferenceForm();
        }
        
        // دالة حذف البدل
        function deleteAllowance(allowanceId) {
            // البحث عن البدل في الجدول
            const row = document.querySelector(`tr[data-allowance-id="${allowanceId}"]`);
            if (row) {
                // حفظ البدل في قائمة البدلات المحذوفة للاستعادة لاحقًا
                const name = row.cells[0].textContent.trim().replace(/\s*×\s*$/, ''); // إزالة علامة الحذف
                const inputs = row.querySelectorAll('input');
                
                if (inputs.length >= 2) {
                    const oldValue = Number(inputs[0].value) || 0;
                    const newValue = Number(inputs[1].value) || 0;
                    
                    // إضافة البدل إلى قائمة البدلات المحذوفة
                    deletedAllowances.push({
                        id: allowanceId,
                        name: name,
                        oldValue: oldValue,
                        newValue: newValue
                    });
                    
                    // حذف البدل من الجدول
                    row.remove();
                    
                    // إضافة زر استعادة البدلات المحذوفة إذا لم يكن موجودًا
                    addRestoreDeletedButton();
                    
                    // عرض رسالة تأكيد
                    const notification = document.createElement('div');
                    notification.className = 'fixed bottom-4 left-4 bg-green-600 text-white p-3 rounded-lg shadow-lg z-50';
                    notification.innerHTML = `تم حذف البدل "${name}" بنجاح`;
                    document.body.appendChild(notification);
                    setTimeout(() => notification.remove(), 3000);
                }
            }
        }

        // استعادة بدل محذوف
        function restoreAllowance(allowanceId) {
            // البحث عن البدل في قائمة البدلات المحذوفة
            const allowanceIndex = deletedAllowances.findIndex(a => a.id === allowanceId);
            if (allowanceIndex !== -1) {
                const allowance = deletedAllowances[allowanceIndex];
                
                // إضافة البدل إلى الجدول مرة أخرى
                const tableBody = document.querySelector('#allowances-table tbody');
                if (!tableBody) {
                    console.error("لم يتم العثور على جدول البدلات");
                    return;
                }
                
                const newRow = document.createElement('tr');
                newRow.setAttribute('data-allowance-id', allowance.id);
                newRow.innerHTML = `
                    <td class="relative">
                        ${allowance.name}
                        <button type="button" class="absolute top-1 left-1 text-red-600 hover:text-red-800 p-1" 
                                onclick="deleteAllowance('${allowance.id}')">
                            <svg xmlns="http://www.w3.org/2000/svg" class="h-4 w-4" fill="none" viewBox="0 0 24 24" stroke="currentColor">
                                <path stroke-linecap="round" stroke-linejoin="round" stroke-width="2" d="M19 7l-.867 12.142A2 2 0 0116.138 21H7.862a2 2 0 01-1.995-1.858L5 7m5 4v6m4-6v6m1-10V4a1 1 0 00-1-1h-4a1 1 0 00-1 1v3M4 7h16" />
                            </svg>
                        </button>
                    </td>
                    <td><input type="number" class="w-full p-2 bg-gray-50 dark:bg-gray-700 border border-gray-300 dark:border-gray-600 rounded-md text-base" value="${allowance.oldValue}" min="0"></td>
                    <td><input type="number" class="w-full p-2 bg-gray-50 dark:bg-gray-700 border border-gray-300 dark:border-gray-600 rounded-md text-base" value="${allowance.newValue}" min="0"></td>
                `;
                tableBody.appendChild(newRow);
                
                // حذف البدل من قائمة البدلات المحذوفة
                deletedAllowances.splice(allowanceIndex, 1);
                
                // تحديث زر الاستعادة أو حذفه إذا لم تعد هناك بدلات محذوفة
                if (deletedAllowances.length === 0) {
                    const restoreBtn = document.getElementById('restore-allowances-btn');
                    if (restoreBtn) restoreBtn.remove();
                    
                    // إغلاق النافذة
                    const modal = document.getElementById('restore-allowances-modal');
                    if (modal) modal.remove();
                } else {
                    // تحديث عدد البدلات المحذوفة في زر الاستعادة
                    const restoreBtn = document.getElementById('restore-allowances-btn');
                    if (restoreBtn) {
                        restoreBtn.innerHTML = `

                                <path stroke-linecap="round" stroke-linejoin="round" stroke-width="2" d="M4 4v5h.582m15.356 2A8.001 8.001 0 004.582 9m0 0H9m11 11v-5h-.581m0 0a8.003 8.003 0 01-15.357-2m15.357 2H15" />
                            </svg>
                            استعادة البدلات المحذوفة (${deletedAllowances.length})
                        `;
                    }
                    
                    // تحديث النافذة
                    showRestoreAllowancesModal();
                }
                
                // عرض رسالة تأكيد
                const notification = document.createElement('div');
                notification.className = 'fixed bottom-4 left-4 bg-green-600 text-white p-3 rounded-lg shadow-lg z-50';
                notification.innerHTML = `تم استعادة البدل "${allowance.name}" بنجاح`;
                document.body.appendChild(notification);
                setTimeout(() => notification.remove(), 3000);
            }

        // تحميل وطباعة نموذج للشخص المحدد من القائمة
        function loadAndPrintPersonForm(personIndex, formType) {
            if (personIndex < 0 || personIndex >= peopleList.length) {
                alert('لم يتم العثور على بيانات الشخص المطلوب');
                return;
            }
            
            const person = peopleList[personIndex];
            
            // تخزين البيانات الحالية لاستعادتها لاحقًا
            const previousData = { ...calculatedData };
            
            // إعداد بيانات محسوبة للطباعة السريعة
            const quickData = {
                name: person.name,
                generalNumber: person.generalNumber,
                newRank: person.rank,
                oldRank: getOldRank(person.rank),
                unit: person.unit || calculatedData.unit || "كتيبة الشرطة العسكرية الحادية عشر",
                idNumber: '',
                startDate: new Date().toISOString().split('T')[0],
                endDate: new Date().toISOString().split('T')[0],
                commencementDate: '',
                daysDiff: 30,
                oldDegree: '1',
                newDegree: '1',
                differences: [
                    { type: 'الراتب الأساسي', oldAmount: 0, newAmount: 0, days: 30, diffAmount: 0 }
                ],
                deductions: [
                    { type: 'التقاعد', oldAmount: 0, newAmount: 0, days: 30, diffAmount: 0 }
                ],
                totalPositiveDiff: 0,
                totalNegativeDiff: 0,
                netDiff: 0
            };
            
            // استبدال بيانات الحساب بالبيانات المؤقتة
            calculatedData = quickData;
            
            // إزالة مربع الحوار
            const modal = document.querySelector('.fixed.inset-0.bg-black.bg-opacity-50');
            if (modal) modal.remove();
            
            // تحديث النموذج المرئي وإعداد نسخة الطباعة
            if (formType === 'difference') {
                previewDifferenceForm();
                setTimeout(() => {
                    // تحديث نسخة الطباعة مباشرة
                    const printDifferenceForm = document.getElementById('print-difference-form');
                    if (printDifferenceForm) {
                        // إظهار نموذج الطباعة وإخفاء البقية
                        document.querySelectorAll('.print-page').forEach(form => {
                            form.classList.add('hidden');
                        });
                        printDifferenceForm.classList.remove('hidden');
                        
                        // تأكد من أن النموذج يحتوي على البيانات المحدثة
                        printDifferenceForm.innerHTML = document.getElementById('difference-form').innerHTML;
                    }
                    
                    window.print();
                    // إعادة البيانات الأصلية
                    calculatedData = previousData;
                }, 500);
            } else if (formType === 'declaration') {
                previewDeclarationForm();
                setTimeout(() => {
                    // تحديث نسخة الطباعة مباشرة
                    const printDeclarationForm = document.getElementById('print-declaration-form');
                    if (printDeclarationForm) {
                        // إظهار نموذج الطباعة وإخفاء البقية
                        document.querySelectorAll('.print-page').forEach(form => {
                            form.classList.add('hidden');
                        });
                        printDeclarationForm.classList.remove('hidden');
                        
                        // تأكد من أن النموذج يحتوي على البيانات المحدثة
                        printDeclarationForm.innerHTML = document.getElementById('declaration-form').innerHTML;
                    }
                    
                    window.print();
                    // إعادة البيانات الأصلية
                    calculatedData = previousData;
                }, 500);
            }
        }
        
        // استنتاج الرتبة السابقة من الرتبة الحالية
        function getOldRank(currentRank) {
            const ranks = ["جندي", "جندي أول", "عريف", "وكيل رقيب", "رقيب", "رقيب أول", "رئيس رقباء", "ملازم"];
            const index = ranks.indexOf(currentRank);
            
            if (index <= 0) return "جندي";
            return ranks[index - 1];
        }

        // الحصول على بيانات سلم الرواتب
        function getSalaryScalesData() {
            const scalesData = {};
            const ranks = ["جندي", "جندي أول", "عريف", "وكيل رقيب", "رقيب", "رقيب أول", "رئيس رقباء", "ملازم"];
            
            // جمع بيانات سلم الرواتب للرتب والدرجات
            ranks.forEach(rank => {
                scalesData[rank] = {};
                for (let degree = 1; degree <= 15; degree++) {
                    const scaleId = `scale-${rank}-${degree}`;
                    const scaleElement = document.getElementById(scaleId);
                    if (scaleElement) {
                        scalesData[rank][degree] = Number(scaleElement.value);
                    }
                }
            });
            
            return scalesData;
        }

        // استيراد البيانات من ملف JSON
        function importDatabase(event) {
            const file = event.target.files[0];
            if (!file) return;
            
            const reader = new FileReader();
            reader.onload = function(e) {
                try {
                    const data = JSON.parse(e.target.result);
                    
                    // التحقق من صحة بنية البيانات
                    if (!data.peopleList || !data.salaryScales) {
                        throw new Error("بنية البيانات غير صحيحة");
                    }
                    
                    // استيراد قائمة الأشخاص
                    peopleList = data.peopleList;
                    updateNamesList();
                    
                    // استيراد سلم الرواتب
                    importSalaryScales(data.salaryScales);
                    
                    // استيراد السجلات المحسوبة إذا كانت موجودة
                    if (data.calculatedRecords && data.calculatedRecords.length > 0) {
                        calculatedData = data.calculatedRecords[0];
                    }
                    
                    // عرض رسالة نجاح
                    const notification = document.createElement('div');
                    notification.className = 'fixed bottom-4 left-4 bg-green-600 text-white p-3 rounded-lg shadow-lg z-50';
                    notification.innerHTML = 'تم استيراد البيانات بنجاح';
                    document.body.appendChild(notification);
                    setTimeout(() => {
                        notification.remove();
                    }, 3000);
                    
                } catch (error) {
                    console.error("خطأ في استيراد البيانات:", error);
                    alert("حدث خطأ أثناء استيراد البيانات: " + error.message);
                }
                
                // إعادة تعيين حقل الملف
                event.target.value = "";
            };
            
            reader.readAsText(file);
        }
        
        // استيراد بيانات سلم الرواتب
        function importSalaryScales(salaryScales) {
            // تحديث قيم سلم الرواتب من البيانات المستوردة
            Object.keys(salaryScales).forEach(rank => {
                Object.keys(salaryScales[rank]).forEach(degree => {
                    const scaleId = `scale-${rank}-${degree}`;
                    const scaleElement = document.getElementById(scaleId);
                    if (scaleElement) {
                        scaleElement.value = salaryScales[rank][degree];
                    }
                });
            });
        }

        // حفظ النموذج الحالي
        function saveCurrentTemplate() {
            if (!editor) {
                alert('المحرر غير متاح حاليًا');
                return;
            }
            
            const templateNameInput = document.getElementById('template-name');
            if (!templateNameInput) {
                console.error("لم يتم العثور على حقل اسم النموذج");
                return;
            }
            
            const templateName = templateNameInput.value.trim();
            if (!templateName) {
                alert('يرجى إدخال اسم للنموذج');
                return;
            }
            
            // الحصول على محتوى النموذج
            const templateContent = editor.getContent();
            
            // إنشاء كائن النموذج
            const template = {
                templateName: templateName,
                templateType: 'custom',
                templateContent: templateContent,
                createdAt: new Date().toISOString()
            };
            
            // حفظ النموذج في التخزين المحلي
            try {
                // الحصول على النماذج المخصصة المخزنة
                let customTemplates = [];
                try {
                    const savedTemplates = localStorage.getItem('customTemplates');
                    if (savedTemplates) {
                        customTemplates = JSON.parse(savedTemplates);
                    }
                } catch (e) {
                    console.error("Error reading custom templates from localStorage:", e);
                    customTemplates = [];
                }
                
                // إضافة النموذج الجديد أو تحديث النموذج الموجود
                const existingIndex = customTemplates.findIndex(t => t.templateName === templateName);
                if (existingIndex !== -1) {
                    // تحديث النموذج الموجود
                    const confirmOverwrite = confirm(`النموذج "${templateName}" موجود بالفعل. هل ترغب في استبداله؟`);
                    if (confirmOverwrite) {
                        customTemplates[existingIndex] = template;
                    } else {
                        return;
                    }
                } else {
                    // إضافة نموذج جديد
                    customTemplates.push(template);
                }
                
                // حفظ النماذج المحدثة في التخزين المحلي
                localStorage.setItem('customTemplates', JSON.stringify(customTemplates));
                
                // عرض رسالة تأكيد
                const notification = document.createElement('div');
                notification.className = 'fixed bottom-4 left-4 bg-green-600 text-white p-3 rounded-lg shadow-lg z-50';
                notification.innerHTML = `تم حفظ النموذج "${templateName}" بنجاح`;
                document.body.appendChild(notification);
                setTimeout(() => notification.remove(), 3000);
                
                // تحديث قائمة النماذج في تبويب النماذج
                addCustomTemplateToForms(template);
            } catch (e) {
                console.error("Error saving template to localStorage:", e);
                
                // تصدير النموذج كملف بديل عن الحفظ في localStorage
                const templateStr = JSON.stringify(template, null, 2);
                const blob = new Blob([templateStr], { type: 'application/json' });
                const url = URL.createObjectURL(blob);
                
                const a = document.createElement('a');
                a.download = `${templateName.replace(/\s+/g, '_')}.json`;
                a.href = url;
                a.style.display = 'none';
                document.body.appendChild(a);
                
                // عرض رسالة للمستخدم
                alert('لا يمكن حفظ النموذج في المتصفح. سيتم تنزيله كملف بدلاً من ذلك.');
                
                a.click();
                
                setTimeout(() => {
                    document.body.removeChild(a);
                    URL.revokeObjectURL(url);
                }, 100);
            }
        }
